// Eine feste ID für den Sender const STREAM_ID = 'mein-wlan-kamera-stream-xyz'; let peer; const videoElement = document.getElementById('video-element'); const statusText = document.getElementById('status'); const roleSelection = document.getElementById('role-selection'); // Hilfsfunktion, um Fehler direkt auf dem Display anzuzeigen function showDebug(msg) { statusText.innerHTML += "
" + msg + ""; console.error(msg); } function initPeer(id = null) { try { peer = id ? new Peer(id) : new Peer(); peer.on('error', (err) => { showDebug("PeerJS Fehler Typ: " + err.type + " - " + err.message); }); } catch(e) { showDebug("Fehler bei PeerJS-Initialisierung: " + e.message); } } // --- ROLLE 1: SENDER (KAMERA) --- function startAsBroadcaster() { roleSelection.classList.add('hidden'); statusText.innerText = "Kamera wird angefordert..."; if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) { showDebug("Kamera-API (getUserMedia) wird von diesem Browser NICHT unterstützt!"); return; } initPeer(STREAM_ID); navigator.mediaDevices.getUserMedia({ video: { facingMode: "environment" }, audio: false }) .then((stream) => { videoElement.srcObject = stream; statusText.innerText = "Kamera aktiv. Warte auf Zuschauer..."; peer.on('call', (call) => { call.answer(stream); statusText.innerText = "Zuschauer verbunden!"; }); }) .catch((err) => { showDebug("Kamera-Zugriff fehlgeschlagen: " + err.name + " - " + err.message); }); } // --- ROLLE 2: EMPFÄNGER (ZUSCHAUER) --- function startAsViewer() { roleSelection.classList.add('hidden'); statusText.innerText = "Verbinde mit Peer-Netzwerk..."; initPeer(); peer.on('open', () => { statusText.innerText = "Rufe Kamera-Handy ab..."; const call = peer.call(STREAM_ID, null); if(!call) { showDebug("Anruf konnte nicht gestartet werden."); return; } call.on('stream', (remoteStream) => { statusText.innerText = "Live-Übertragung aktiv"; videoElement.srcObject = remoteStream; }); // Falls der Verbindungspartner auflegt oder wegbricht call.on('error', (err) => { showDebug("Verbindungsfehler im Call: " + err.message); }); }); }
